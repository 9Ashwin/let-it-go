```go
var thresholdsPath = "inventory/thresholds.json"   // 包内测试接缝

func LowStockThreshold(warehouse string) int {
    if threshold, ok := readThresholds(thresholdsPath)[warehouse]; ok {
        return threshold
    }
    return DefaultLowStockThreshold
}
```
